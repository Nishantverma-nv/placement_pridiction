    train, test = pd.read_csv("data/preprocessed_train.csv"), pd.read_csv("data/preprocessed_test.csv")
    for name, model in make_models().items():
        model.fit(Xtr, ytr)                        # fit on the training fold only
        metrics[name] = evaluate(yte, model.predict(Xte), model.predict_proba(Xte))
        joblib.dump(model, MODELS_DIR / f"{name.lower()}.pkl")
    best = max(models, key=lambda n: metrics[n]["f1"])     # deployment rule: best F1
    joblib.dump(models[best], DEFAULT_MODEL)
